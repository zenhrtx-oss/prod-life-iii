## The script

```lua
-- Scratch: today's pacing_spec inputs and policies over a chosen seed set (default: 100000 + 7919*i, i = 0..199).
-- Mirrors spec/pacing_spec.lua's run(), ODDS, GATES, stats helpers. Arg "spec" runs CareerSim.Seeds instead (control).
local Sim = require("@ok/src/ReplicatedStorage/OK/Modules/CareerSim")
local R = require("@ok/src/ReplicatedStorage/OK/Modules/WalkInRules")
local CR = require("@ok/src/ReplicatedStorage/OK/Modules/CareerRules")
local W = require("@ok/src/ReplicatedStorage/OK/Data/WalkIns")
local E = require("@ok/spec/economy_fixture")

local ODDS = { auraPer = 20, auraCap = 15, followersPer = 1000, followersCap = 10, min = 10, max = 90 }
local GATES = {
	studio_two = { needs = { placements = 1, aura = 300 }, cost = 5000 },
	plate_legend = { needs = { hits = 1, followers = 20000, aura = 1000 }, cost = 4000, level = 10 },
}
local DAYS = 60
local function run(seed, policy, spendDown)
	return Sim.Run({ days = DAYS, economy = E, deck = W.Deck, rules = R, careerRules = CR, odds = ODDS, followupAt = { 2, 4 },
		dropAt = { 1, 1 }, cooldown = 3, walkinsPerDay = 8, phonePerDay = 6, phoneSeed = seed, auraHeal = 5, rng = Sim.Lcg(seed),
		policy = policy, spendDown = spendDown, gates = GATES })
end
local function sorted(values) local s = {} for i, v in ipairs(values) do s[i] = v end table.sort(s) return s end
local function median(values)
	local s, n = sorted(values), #values
	if n % 2 == 1 then return s[(n + 1) // 2] end
	return (s[n // 2] + s[n // 2 + 1]) / 2
end
local function percentile(values, k) local s = sorted(values) return s[math.max(1, math.ceil(k / 100 * #s))] end
local function dayOrInf(d) return d or math.huge end
local function fmt(x) return x == math.huge and "never" or string.format("%g", x) end

local seeds = {}
if (...) == "spec" then seeds = Sim.Seeds else for i = 0, 199 do seeds[#seeds + 1] = 100000 + 7919 * i end end
local t0 = os.clock()

local twoDays, shares, legend, legendMin, pops, aura = {}, {}, {}, math.huge, 0, 0
local fails = {}
local function note(k, s) fails[k] = fails[k] or {} table.insert(fails[k], s) end
for _, seed in ipairs(seeds) do
	local res = run(seed, "random")
	table.insert(twoDays, dayOrInf(res.studioTwoDay))
	table.insert(shares, res.steadyShare)
	table.insert(legend, dayOrInf(res.legendDay))
	legendMin = math.min(legendMin, dayOrInf(res.legendDay))
	pops += res.phonePops / DAYS / #seeds
	aura += res.phoneAura / DAYS / #seeds
	if not (res.legendDay == nil or res.legendDay >= 15) then note("legend random", seed .. ":" .. res.legendDay) end
	if res.steadyShare > 0.50 then note("ceiling random", string.format("%d:%.4f", seed, res.steadyShare)) end
end
local function policyMax(policy, spendDown, label)
	local mx, lmin, sh = 0, math.huge, {}
	for _, seed in ipairs(seeds) do
		local res = run(seed, policy, spendDown)
		mx = math.max(mx, res.steadyShare)
		table.insert(sh, res.steadyShare)
		lmin = math.min(lmin, dayOrInf(res.legendDay))
		if res.steadyShare > 0.50 then note("ceiling " .. label, string.format("%d:%.4f", seed, res.steadyShare)) end
	end
	return mx, lmin, median(sh)
end
local gMax, gLeg, gMed = policyMax("greedy", nil, "greedy")
local cMax, cLeg, cMed = policyMax("chaser", nil, "chaser")
local sMax, sLeg, sMed = policyMax("random", true, "spend-down")
local elapsed = os.clock() - t0

local twoMed, twoP10, twoP90 = median(twoDays), percentile(twoDays, 10), percentile(twoDays, 90)
local hist = {}
for _, d in ipairs(twoDays) do local k = fmt(d) hist[k] = (hist[k] or 0) + 1 end
local hk = {} for k in pairs(hist) do hk[#hk + 1] = k end table.sort(hk, function(a, b) return (tonumber(a) or 1e9) < (tonumber(b) or 1e9) end)
local hs = {} for _, k in ipairs(hk) do hs[#hs + 1] = k .. "=" .. hist[k] end

print(string.format("seeds %d (first %d, last %d)", #seeds, seeds[1], seeds[#seeds]))
print(string.format("random share median %.4f (mean of sorted %d/%d: %.6f / %.6f); min %.4f max %.4f",
	median(shares), #shares // 2, #shares // 2 + 1, sorted(shares)[#shares // 2], sorted(shares)[#shares // 2 + 1], sorted(shares)[1], sorted(shares)[#shares]))
print(string.format("greedy max %.4f (median %.4f, legend min %s)", gMax, gMed, fmt(gLeg)))
print(string.format("chaser max %.4f (median %.4f, legend min %s)", cMax, cMed, fmt(cLeg)))
print(string.format("spend-down max %.4f (median %.4f, legend min %s)", sMax, sMed, fmt(sLeg)))
print(string.format("THE HALL (studio_two) random: median %s p10 %s p90 %s; days: %s", fmt(twoMed), fmt(twoP10), fmt(twoP90), table.concat(hs, " ")))
print(string.format("legend random: earliest %s, median %s, p10 %s", fmt(legendMin), fmt(median(legend)), fmt(percentile(legend, 10))))
print(string.format("phone pops/day %.4f; phone aura/day %.4f", pops, aura))
print(string.format("guardrails: HALL median 4..6 %s; p10>=3 %s; p90<=7 %s; legend>=15 %s; ceiling<=0.50 every seed %s; random median<=0.50 %s; aura<=15 %s",
	tostring(twoMed >= 4 and twoMed <= 6), tostring(twoP10 >= 3), tostring(twoP90 <= 7), tostring(fails["legend random"] == nil),
	tostring(not (fails["ceiling random"] or fails["ceiling greedy"] or fails["ceiling chaser"] or fails["ceiling spend-down"])),
	tostring(median(shares) <= 0.50), tostring(aura <= 15)))
for k, v in pairs(fails) do print("FAIL " .. k .. " (" .. #v .. " seeds): " .. table.concat(v, ", ")) end
print(string.format("cpu %.2f s", elapsed))
```
