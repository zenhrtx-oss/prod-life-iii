local function freshRequire(mod) -- edit-mode require caches modules across runs; load a clone so edits take effect
  local c = mod:Clone(); c.Parent = mod.Parent; c.Name = "_fresh_" .. mod.Name
  local ok, res = pcall(require, c); c:Destroy(); if not ok then error(res) end; return res
end

local ENGINE = [===[
-- OK Beat Maker playback engine (client). Steps fire on Heartbeat; late steps are dropped instead of bunching up.
local RunService, Debris, SoundService = game:GetService("RunService"), game:GetService("Debris"), game:GetService("SoundService")
local E = {}; E.__index = E
function E.new(cfg, samples) return setmetatable({cfg = cfg, samples = samples, playing = false, step = 0, nextT = 0, hits = 0}, E) end
function E:Trigger(t)
  local tr = self.cfg.Tracks[t]; local tmpl = tr and self.samples:FindFirstChild(tr.Name); if not tmpl then return end
  local s = tmpl:Clone(); s.Parent = SoundService; s:Play(); Debris:AddItem(s, tr.Gate)
  self.hits += 1
end
function E:Start(getPattern, getBPM, onStep)
  if self.playing then return end
  self.playing = true; self.step = 0; self.nextT = os.clock()
  self.conn = RunService.Heartbeat:Connect(function()
    local now = os.clock(); if now < self.nextT then return end
    local interval = 60 / getBPM() / 4
    if now - self.nextT > interval then self.nextT = now end -- fell behind: drop late steps
    self.nextT += interval
    self.step = self.step % self.cfg.Steps + 1
    local p, n = getPattern(), self.cfg.Steps
    for t = 1, #self.cfg.Tracks do local i = (t - 1) * n + self.step; if p:sub(i, i) == "1" then self:Trigger(t) end end
    if onStep then onStep(self.step, self.hits) end
  end)
end
function E:Stop() self.playing = false; if self.conn then self.conn:Disconnect(); self.conn = nil end; self.step = 0 end
return E
]===]

local SYSTEM = [===[
-- OK Studio system: DAW (OK Beat Maker v0). The studio PC opens the beat maker; saves are validated here.
local RS, Players = game:GetService("ReplicatedStorage"), game:GetService("Players")
local Cfg = require(RS.OK.Modules.DAWConfig)
local M = { Enabled = true, Version = 1 }
local lastSave, lastXP = {}, {}
function M.Validate(pattern, bpm)
  if type(pattern) ~= "string" or #pattern ~= #Cfg.Tracks * Cfg.Steps or pattern:find("[^01]") then return nil end
  if type(bpm) ~= "number" or bpm ~= bpm or bpm == math.huge or bpm == -math.huge then return nil end
  return pattern, math.clamp(math.floor(bpm + 0.5), Cfg.BPM.Min, Cfg.BPM.Max)
end
function M.Init(ctx)
  RS.OK.Remotes.DAW_Save.OnServerInvoke = function(player, pattern, bpm)
    local p, b = M.Validate(pattern, bpm)
    if not p then return { ok = false, reason = "invalid" } end
    local now = os.clock()
    if lastSave[player] and now - lastSave[player] < Cfg.SaveCooldown then return { ok = false, reason = "cooldown" } end
    lastSave[player] = now
    local prof = ctx.Data.GetProfile(player)
    prof.daw = { pattern = p, bpm = b, saves = ((prof.daw and prof.daw.saves) or 0) + 1 }
    local xp = 0
    if not lastXP[player] or now - lastXP[player] >= Cfg.XPCooldown then lastXP[player] = now; ctx.Data.AddXP(player, Cfg.XPPerSave); xp = Cfg.XPPerSave end
    print(string.format("[OK] %s saved a beat (%d BPM, save #%d, +%d XP)", player.Name, b, prof.daw.saves, xp))
    return { ok = true, xp = xp, profile = prof }
  end
  ctx.Interaction.Register("Computer", function(player, inst)
    local prof = ctx.Data.GetProfile(player)
    RS.OK.Remotes.InteractionEvent:FireClient(player, "OpenDAW", { source = inst:GetFullName(), daw = prof.daw })
  end)
  Players.PlayerRemoving:Connect(function(p) lastSave[p] = nil; lastXP[p] = nil end)
end
return M
]===]

local CLIENT = [===[
-- OK Beat Maker client: builds the window, plays the samples locally, saves through the server.
local RS, Players, RunService, CP = game:GetService("ReplicatedStorage"), game:GetService("Players"), game:GetService("RunService"), game:GetService("ContentProvider")
local player = Players.LocalPlayer
local Cfg = require(RS.OK.Modules.DAWConfig); local Theme = require(RS.OK.Modules.UITheme)
local Engine = require(RS.OK.Modules.DAWEngine); local UI = require(RS.OK.Modules.DAWUI)
local Event, SaveFn = RS.OK.Remotes.InteractionEvent, RS.OK.Remotes:WaitForChild("DAW_Save")
local samples = RS.OK:WaitForChild("Samples")
task.spawn(function() pcall(function() CP:PreloadAsync(samples:GetChildren()) end) end)
local gui = Instance.new("ScreenGui"); gui.Name = "OK_DAW"; gui.ResetOnSpawn = false; gui.DisplayOrder = 5; gui.Parent = player:WaitForChild("PlayerGui")
local ui = UI.Build(gui, Theme, Cfg); local root = ui.Root
local engine = Engine.new(Cfg, samples)
local sourcePos = nil
ui.OnPlayChanged = function(on)
  if on then
    engine:Start(function() return root:GetAttribute("Pattern") end, function() return root:GetAttribute("BPM") end, function(step, hits) root:SetAttribute("Step", step); root:SetAttribute("Hits", hits) end)
  else engine:Stop(); root:SetAttribute("Step", 0) end
  player:SetAttribute("DAWPlaying", on)
end
ui.OnSave = function()
  root:SetAttribute("Status", "Saving…")
  local ok, res = pcall(function() return SaveFn:InvokeServer(root:GetAttribute("Pattern"), root:GetAttribute("BPM")) end)
  if ok and type(res) == "table" and res.ok then
    root:SetAttribute("Status", res.xp > 0 and ("Saved · +" .. res.xp .. " XP") or "Saved"); root:SetAttribute("LastSave", "ok")
  else
    local reason = (ok and type(res) == "table" and res.reason) or "error"
    root:SetAttribute("Status", reason == "cooldown" and "Wait a moment before saving again" or ("Not saved (" .. reason .. ")")); root:SetAttribute("LastSave", reason)
  end
end
local function resolve(fullName) local node = game; for seg in string.gmatch(fullName, "[^%.]+") do node = node and node:FindFirstChild(seg) end; return node end
Event.OnClientEvent:Connect(function(kind, payload)
  if kind ~= "OpenDAW" then return end
  local src = payload.source and resolve(payload.source); sourcePos = (src and src:IsA("BasePart")) and src.Position or nil
  local d = payload.daw
  ui.Show((d and d.pattern) or Cfg.DefaultPattern, (d and d.bpm) or Cfg.BPM.Default)
end)
RunService.Heartbeat:Connect(function()
  if ui.IsOpen() and sourcePos then
    local r = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if r and (r.Position - sourcePos).Magnitude > Cfg.CloseDistance then ui.Hide() end
  end
end)
