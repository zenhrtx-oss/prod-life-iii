src(ensure(Services, "ModuleScript", "DataService"), [==[
local DSS, Players = game:GetService("DataStoreService"), game:GetService("Players")
local Config = require(game:GetService("ReplicatedStorage").OK.Config)
local M = {}; local store; local cache = {}
local DEFAULT = { level = 1, visits = 0, xp = 0 }
local function key(p) return "u_" .. p.UserId end
function M.Load(p)
  local prof = table.clone(DEFAULT)
  if store then local ok, data = pcall(function() return store:GetAsync(key(p)) end); if ok and type(data) == "table" then for k, v in pairs(data) do prof[k] = v end end end
  prof.visits += 1; cache[p.UserId] = prof; return prof
end
function M.GetProfile(p) return cache[p.UserId] or M.Load(p) end
function M.Save(p) local prof = cache[p.UserId]; if store and prof then pcall(function() store:SetAsync(key(p), prof) end) end end
function M.AddXP(p, n) local prof = M.GetProfile(p); prof.xp += n; prof.level = 1 + math.floor(prof.xp / Config.XP.LevelSize); return prof end
function M.Init()
  local ok, s = pcall(function() return DSS:GetDataStore(Config.DataStoreName) end); store = ok and s or nil
  Players.PlayerAdded:Connect(function(p) M.Load(p) end)
  Players.PlayerRemoving:Connect(function(p) M.Save(p); cache[p.UserId] = nil end)
  for _, p in ipairs(Players:GetPlayers()) do M.Load(p) end
  game:BindToClose(function() for _, p in ipairs(Players:GetPlayers()) do M.Save(p) end end)
end
return M
]==])

src(ensure(Services, "ModuleScript", "LightingService"), [==[
local Lighting, TS = game:GetService("Lighting"), game:GetService("TweenService")
local Config = require(game:GetService("ReplicatedStorage").OK.Config)
local M = {}; M.Current = "Warm"
local function tween(inst, props, t) TS:Create(inst, TweenInfo.new(t or 1), props):Play() end
local function isStrip(n) return n:match("^Strip_") or n == "Desk_LED" or n == "Bar_LED" end
local function isPendant(n) return (n:match("_Bulb$") or n:match("_Lens$")) and n ~= "BoothLamp" end
function M.Apply(name)
  local p = Config.LightPresets[name]; if not p then return false end
  M.Current = name
  tween(Lighting, {Ambient = p.Ambient, ExposureCompensation = p.Exposure}, 1)
  for _, d in ipairs(workspace.OKStudio:GetDescendants()) do
    if d:IsA("BasePart") and d.Material == Enum.Material.Neon then
      if isPendant(d.Name) then tween(d, {Color = p.Pendant}, 1) elseif isStrip(d.Name) then tween(d, {Color = p.Strip}, 1) end
      local l = d:FindFirstChildWhichIsA("Light")
      if l then if isPendant(d.Name) then tween(l, {Color = p.Pendant}, 1) elseif isStrip(d.Name) then tween(l, {Color = p.Strip}, 1) end end
    end
  end
  workspace.OKStudio:SetAttribute("LightPreset", name)
  return true
end
function M.Toggle() M.Apply(M.Current == "Warm" and "Neon" or "Warm"); return M.Current end
function M.Init() workspace.OKStudio:SetAttribute("LightPreset", M.Current) end
return M
]==])
